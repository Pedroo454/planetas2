function normalizeBirthday(raw){
  const match = String(raw || '').trim().match(/^\s*(\d{1,2})\s*[/.-]\s*(\d{1,2})\s*$/);
  if (!match) throw new Error('Digite a data no formato DD/MM, por exemplo 30/07.');
  const day = Number(match[1]), month = Number(match[2]);
  const maxDays = [31,29,31,30,31,30,31,31,30,31,30,31][month-1];
  if (!maxDays || day < 1 || day > maxDays) throw new Error('Essa data não é válida. Confira o dia e o mês.');
  return `${String(day).padStart(2,'0')}/${String(month).padStart(2,'0')}`;
}

// Leitor CSV que respeita campos entre aspas e aspas escapadas (""), inclusive descrições com vírgulas.
function parseCsv(text){
  const rows = [];
  let row = [], field = '', quoted = false;
  for (let i = 0; i < text.length; i++){
    const ch = text[i];
    if (quoted){
      if (ch === '"' && text[i+1] === '"'){ field += '"'; i++; }
      else if (ch === '"') quoted = false;
      else field += ch;
    } else {
      if (ch === '"') quoted = true;
      else if (ch === ','){ row.push(field); field = ''; }
      else if (ch === '\n'){
        row.push(field); field = '';
        if (row.some(value => value.trim() !== '')) rows.push(row);
        row = [];
      } else if (ch !== '\r') field += ch;
    }
  }
  if (field.length || row.length){ row.push(field); if (row.some(value => value.trim() !== '')) rows.push(row); }
  if (rows.length < 2) throw new Error('A base Hubble30 está vazia ou em formato inválido.');
  const headers = rows.shift().map(h => h.trim());
  return rows.map(values => Object.fromEntries(headers.map((header, i) => [header, (values[i] || '').trim()])));
}

async function loadHubbleDatabase(){
  if (!hubbleDataPromise){
    hubbleDataPromise = fetch(HUBBLE_CSV_URL, {cache:'force-cache'}).then(async response => {
      if (!response.ok) throw new Error('Não foi possível carregar a base Hubble30 (HTTP ' + response.status + ').');
      return parseCsv(await response.text());
    }).catch(error => { hubbleDataPromise = null; throw error; });
  }
  return hubbleDataPromise;
}

function dateKeyFromCsv(value){
  const parsed = new Date(value);
  if (Number.isNaN(parsed.getTime())) return '';
  // A data da base vem no formato inglês "January 1 2019"; só usamos dia e mês.
  return `${String(parsed.getUTCDate()).padStart(2,'0')}/${String(parsed.getUTCMonth()+1).padStart(2,'0')}`;
}
function getRecordForDate(records, date){ return records.find(item => dateKeyFromCsv(item.Date) === date); }
function recordImage(item){ return HUBBLE_IMAGE_BASE + encodeURIComponent(item.Image || ''); }
function recordTitle(item){ return item.Name || 'Objeto celeste do Hubble'; }
function recordDescription(item){ return item.Caption || 'Imagem do arquivo de aniversário do Telescópio Espacial Hubble.'; }
function recordSource(item){ return item.URL || 'https://imagine.gsfc.nasa.gov/docs/hst_bday/'; }

/* ================= FOTOS DOS ASTRONAUTAS =================
   Buscadas automaticamente na Wikipédia/Wikimedia Commons.
   Para usar uma foto sua, coloque o arquivo na mesma pasta do site e preencha "photo" (ex.: photo:'marcos-pontes.jpg'). */
const ASTRO = {
  'Marcos Pontes':   {photo:'', q:'Marcos Pontes astronauta', must:'pontes'},
  'Victor Hespanha': {photo:'', q:'Victor Hespanha', must:'hespanha'},
};
async function wikiPhoto(q, must){
  for (const lang of ['pt', 'en']){
    try{
      const r = await fetch(`https://${lang}.wikipedia.org/w/api.php?action=query&format=json&origin=*&generator=search&gsrsearch=${encodeURIComponent(q)}&gsrlimit=4&prop=pageimages&piprop=thumbnail&pithumbsize=400`);
      const j = await r.json();
      const pages = Object.values(j.query?.pages || {}).sort((a, b) => a.index - b.index);
      const p = pages.find(p => p.thumbnail && p.title.toLowerCase().includes(must));
      if (p) return p.thumbnail.source;
    }catch(e){}
  }
  return '';
}
document.querySelectorAll('.astro-photo').forEach(async el => {
  const c = ASTRO[el.dataset.name]; if (!c) return;
  const src = c.photo || await wikiPhoto(c.q, c.must);
  if (!src) return; // sem foto: fica o avatar com as iniciais
  const im = new Image(); im.alt = 'Foto de ' + el.dataset.name; im.referrerPolicy = 'no-referrer';
  im.onload = () => el.appendChild(im);
  im.src = src;
});

  const fmt = v => v.toLocaleString('pt-BR', {minimumFractionDigits:dec, maximumFractionDigits:dec}) + suf;

  const t0 = performance.now(), dur = 1600;
